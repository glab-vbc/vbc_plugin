# mirrors HOOMD-blue's own requirement (hoomd-blue/CMakeLists.txt)
cmake_minimum_required(VERSION 3.15...3.27)

# name of the plugin
project(vbc_plugin LANGUAGES C CXX)

set(COMPONENT_NAME vbc_plugin)

# find hoomd first
find_package(HOOMD 4.8.0 REQUIRED)

# print where you found it
message(STATUS "Found HOOMD ${HOOMD_VERSION}: ${HOOMD_INSTALL_PREFIX}/${PYTHON_SITE_INSTALL_DIR}")

# find pybind11, needed for python binding
find_package(pybind11 REQUIRED)

# add hoomd path to cmake
set(CMAKE_INSTALL_PREFIX ${HOOMD_INSTALL_PREFIX})

# source files for plugin, .cc
set(_${COMPONENT_NAME}_sources 
    src/module.cc
)

# source files for cuda, .cu
set(_cuda_sources "")

if (ENABLE_HIP)
set(_cuda_sources 
    src/PotentialValencyVBCGPUKernel.cu
)
endif (ENABLE_HIP)

# all the header files
set(_${COMPONENT_NAME}_headers
    src/EvaluatorVBC.h
    src/PotentialValency.h
    src/PotentialValencyGPU.h
)

# python file where potentials python calling is defined
set(py_files src/vbc_potential.py)

# Add the module
hoomd_add_module(_${COMPONENT_NAME} SHARED ${_${COMPONENT_NAME}_sources} ${_cuda_sources} ${_${COMPONENT_NAME}_headers} NO_EXTRAS)

add_library(HOOMD::_${COMPONENT_NAME} ALIAS _${COMPONENT_NAME})

# Now that the target _${COMPONENT_NAME} exists, link HOOMD libraries to it
target_link_libraries(_${COMPONENT_NAME}
    PUBLIC HOOMD::_hoomd
    PUBLIC HOOMD::_md
)

# Include directories for HOOMD and build directory for autogenerated files
target_include_directories(_${COMPONENT_NAME} PRIVATE
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>
    ${HOOMD_INCLUDE_DIRS}
)

set_target_properties(_${COMPONENT_NAME} PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)

message(STATUS "Building _${COMPONENT_NAME} with sources: ${_${COMPONENT_NAME}_sources}")

install(TARGETS _${COMPONENT_NAME}
        LIBRARY DESTINATION ${PYTHON_SITE_INSTALL_DIR}/${COMPONENT_NAME}
        )

# Install Python files
install(FILES ${py_files}
        DESTINATION ${PYTHON_SITE_INSTALL_DIR}/${COMPONENT_NAME}
       )
