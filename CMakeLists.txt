# Add the module
hoomd_add_module(_${COMPONENT_NAME} SHARED ${_${COMPONENT_NAME}_sources} ${_cuda_sources} ${DFFT_SOURCES} ${_${COMPONENT_NAME}_headers} NO_EXTRAS)

add_library(HOOMD::_${COMPONENT_NAME} ALIAS _${COMPONENT_NAME})

# Now that the target _${COMPONENT_NAME} exists, link HOOMD libraries to it
target_link_libraries(_${COMPONENT_NAME}
    PUBLIC HOOMD::_hoomd
    PUBLIC HOOMD::_md
)

# Include directories for HOOMD and build directory for autogenerated files
target_include_directories(_${COMPONENT_NAME} PRIVATE
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>
    ${HOOMD_INCLUDE_DIRS}
)

set_target_properties(_${COMPONENT_NAME} PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)

message(STATUS "Building _${COMPONENT_NAME} with sources: ${_${COMPONENT_NAME}_sources}")

install(TARGETS _${COMPONENT_NAME}
        LIBRARY DESTINATION ${PYTHON_SITE_INSTALL_DIR}/${COMPONENT_NAME}
        )

# Install Python files
install(FILES ${py_files}
        DESTINATION ${PYTHON_SITE_INSTALL_DIR}/${COMPONENT_NAME}
       )
