cmake_minimum_required(VERSION 3.9 FATAL_ERROR)

# name of the plugin
project(sqd_plugin LANGUAGES C CXX)

# find hoomd first
find_package(HOOMD 4.0.0 REQUIRED)

# print where you found it
message(STATUS "Found HOOMD ${HOOMD_VERSION}: ${HOOMD_INSTALL_PREFIX}/${PYTHON_SITE_INSTALL_DIR}")

# find pybind11, needed for python binding
find_package(pybind11 REQUIRED)

#set(ENABLE_HIP ${ENABLE_GPU})

# add hoomd path to cmake
set(CMAKE_INSTALL_PREFIX ${HOOMD_INSTALL_PREFIX})

# source files for plugin, .cc
set(_sqd_plugin_sources "")

# source files for cuda, .cu
set(_cuda_sources "")

# all the header files
set(_sqd_plugin_headers
    src/EvaluatorSquareDensity.h
    src/EvaluatorTersoff.h
    src/PotentialTersoff.h
    src/PotentialTersoffGPU.h
)

if (ENABLE_HIP)
set(_cuda_sources ${_cuda_sources} ${DFFT_CU_SOURCES})
endif (ENABLE_HIP)

# python file where potentials python calling is defined
set(py_files src/sqd_potential.py)


## copy pasted from hoomd cmake file
# Set triplet evaluators
set(_triplets Tersoff SquareDensity)

foreach(_evaluator ${_triplets})
    configure_file(src/export_PotentialTersoff.cc.inc
                   src/export_PotentialTersoff${_evaluator}.cc
                   @ONLY)
    set(_sqd_plugin_sources ${_sqd_plugin_sources} src/export_PotentialTersoff${_evaluator}.cc)

    if (ENABLE_HIP)
        configure_file(src/export_PotentialTersoffGPU.cc.inc
                       src/export_PotentialTersoff${_evaluator}GPU.cc
                       @ONLY)
        configure_file(src/PotentialTersoffGPUKernel.cu.inc
                       src/PotentialTersoff${_evaluator}GPUKernel.cu
                       @ONLY)
        set(_sqd_plugin_sources ${_sqd_plugin_sources} src/export_PotentialTersoff${_evaluator}GPU.cc)
        set(_cuda_sources ${_cuda_sources}
            src/PotentialTersoff${_evaluator}GPUKernel.cu
            )
        set_source_files_properties(${_cuda_sources} PROPERTIES LANGUAGE ${HOOMD_DEVICE_LANGUAGE})
    endif()
endforeach()

# Add the module
hoomd_add_module(_sqd_plugin SHARED ${_sqd_plugin_sources} ${_cuda_sources} ${DFFT_SOURCES} ${_sqd_plugin_headers} NO_EXTRAS)

# Now that the target _sqd_plugin exists, link HOOMD libraries to it
target_link_libraries(_sqd_plugin
    PUBLIC HOOMD::_hoomd
    PUBLIC HOOMD::_md
)

# Include directories for HOOMD and build directory for autogenerated files
target_include_directories(_sqd_plugin PRIVATE
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>
    ${HOOMD_INCLUDE_DIRS}
)

set_target_properties(_sqd_plugin PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)

message(STATUS "Building _sqd_plugin with sources: ${_sqd_plugin_sources}")

# Install Python files
install(FILES ${py_files}
        DESTINATION ${PYTHON_SITE_INSTALL_DIR}/sqd_plugin
       )
